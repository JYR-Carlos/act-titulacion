# Pipeline de IA / datos LSCh-MR.

# Versiones PINEADAS con `==`, no con `>=`, a propósito. La combinación
# tf2onnx + Keras 3 es frágil: una versión más nueva de TensorFlow rompe la
# exportación a ONNX, y el fallo aparece al final del pipeline (al exportar),
# no al instalar. Estas son las versiones con las que se produjo el
# `outputs/models/modelo.onnx` que viaja a Unity.
#
# Entorno de referencia: Python 3.12.0 (ver `.python-version`), Windows 11.
# El pipeline está separado en dos grupos: la CAPTURA/DATASET/INFERENCIA no
# necesita TensorFlow; solo el ENTRENAMIENTO y la EXPORTACIÓN lo requieren.

# --- Núcleo: captura, dataset, inferencia (no requiere TensorFlow) -----------
numpy==2.3.5
opencv-python==4.13.0.92
mediapipe==0.10.35         # Tasks API / HandLandmarker (NO la API legacy)
onnx==1.22.0

scikit-learn==1.7.2
matplotlib==3.10.7
psutil==7.1.3              # MonitorRecursos (demo_vivo.py): CPU/memoria del proceso

# --- Entrenamiento + exportación a ONNX -------------------------------------
# TensorFlow 2.21 trae Keras 3.15 por defecto; el ModelExporter maneja tanto
# Keras 3 como Keras 2 y ofrece varias rutas de exportación (ver
# lsch_mr/model_exporter.py). Si subes tensorflow o tf2onnx, vuelve a correr
# `python exportar_onnx.py` y `python generar_vectores_dorados.py --verificar`
# antes de dar el entorno por bueno: si el ONNX cambia, el port en C# queda
# invalidado (ver INTEGRACION_UNITY.md).

tf2onnx==1.17.0
